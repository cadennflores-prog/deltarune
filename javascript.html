async function downloadDeltaruneSave() { console.log("Starting save search..."); let filesToDownload = []; // 1. Search LocalStorage const keys = Object.keys(localStorage); keys.forEach(key => { if (key.includes('file') || key.includes('dr_') || key.includes('config')) { filesToDownload.push({ name: key, data: localStorage.getItem(key) }); } }); // 2. Search IndexedDB (This is where newer GameMaker builds hide data) try { const dbs = await window.indexedDB.databases(); for (let dbInfo of dbs) { const db = await new Promise(res => { const req = indexedDB.open(dbInfo.name); req.onsuccess = () => res(req.result); }); // Usually "FILE_DATA" or "files" const storeNames = Array.from(db.objectStoreNames); for (let storeName of storeNames) { const tx = db.transaction(storeName, "readonly"); const store = tx.objectStore(storeName); const allKeys = await new Promise(res => { const req = store.getAllKeys(); req.onsuccess = () => res(req.result); }); for (let key of allKeys) { if (typeof key === 'string' && (key.includes('file') || key.includes('dr_'))) { const dataReq = store.get(key); const data = await new Promise(res => { dataReq.onsuccess = () => res(dataReq.result); }); filesToDownload.push({ name: key.replace(/\//g, '_'), data: data }); } } } db.close(); } } catch (e) { console.log("IndexedDB search skipped or failed: ", e); } // 3. Execute Downloads if (filesToDownload.length === 0) { alert("Search finished: 0 save files found. Try saving in-game again first!"); return; } alert("Found " + filesToDownload.length + " files. Starting download..."); for (let file of filesToDownload) { const content = typeof file.data === 'object' ? JSON.stringify(file.data) : file.data; const blob = new Blob([content], { type: 'text/plain' }); const url = window.URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = file.name; document.body.appendChild(a); a.click(); document.body.removeChild(a); window.URL.revokeObjectURL(url); await new Promise(r => setTimeout(r, 300)); // Delay to prevent browser blocking } }
