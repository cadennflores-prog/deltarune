async function downloadDeltaruneSave() { // 1. A more complete list of possible Deltarune save names const saveKeys = [ 'file0', 'file1', 'file2', 'file3', 'file4', 'file5', 'file6', 'file9', 'filech3_0', 'filech3_1', 'filech3_2', 'filech3_3', 'filech4_0', 'filech4_1', 'filech4_2', 'filech4_3', 'dr_ini', 'deltarune_config', 'userconfig', 'keyconfig' ]; let foundAny = false; for (const key of saveKeys) { const data = localStorage.getItem(key); if (data) { foundAny = true; // Create the download link const blob = new Blob([data], { type: 'text/plain' }); const url = window.URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = key; // No extension, just like Deltarune likes it document.body.appendChild(a); a.click(); // Cleanup window.URL.revokeObjectURL(url); document.body.removeChild(a); // 2. IMPORTANT: Add a tiny delay between downloads // Some browsers block "mass downloads" if they happen at the exact same millisecond. await new Promise(resolve => setTimeout(resolve, 200)); } } if (!foundAny) { alert("Search complete: No save data found in this browser's LocalStorage. Are you sure you've saved in-game?"); } }
